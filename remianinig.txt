from langgraph.graph import END, StateGraph
from langchain_aws import ChatBedrockConverse
from langchain_core.messages import HumanMessage
from typing import TypedDict, Literal
class StateSchema(TypedDict):
    topic: str
    response: str
    is_biased: bool
    regenerated: bool
def generate_response(state: StateSchema) -> StateSchema:
    print(f"\nGenerating response for: {state['topic']}")
    llm = ChatBedrockConverse(model="amazon.nova-lite-v1:0", temperature=0.7)
    prompt = f"Provide a detailed response about: {state['topic']}"
    response = llm.invoke([HumanMessage(content=prompt)])
    return {
        "topic": state["topic"],
        "response": response.content,
        "is_biased": False,
        "regenerated": state.get("regenerated", False)
    }
def validate_bias(state: StateSchema) -> StateSchema:
    print("Checking for bias...")
    llm = ChatBedrockConverse(model="amazon.nova-lite-v1:0", temperature=0)
    prompt = f
    result = llm.invoke([HumanMessage(content=prompt)])
    bias_found = "yes" in result.content.lower()
    print(f"Biased: {bias_found}\n")
    return {
        "topic": state["topic"],
        "response": state["response"],
        "is_biased": bias_found,
        "regenerated": state.get("regenerated", False)
    }
def regenerate_unbiased(state: StateSchema) -> StateSchema:
    print("Regenerating unbiased content...")
    llm = ChatBedrockConverse(model="amazon.nova-lite-v1:0", temperature=0.7)
    prompt = f
    response = llm.invoke([HumanMessage(content=prompt)])
    return {
        "topic": state["topic"],
        "response": response.content,
        "is_biased": False,
        "regenerated": True
    }
def route_bias(state: StateSchema) -> Literal["regenerate", "output"]:
    if state["is_biased"] and not state.get("regenerated", False):
        return "regenerate"
    return "output"
def show_output(state: StateSchema) -> StateSchema:
    print("="*60)
    print(f"Topic: {state['topic']}")

    print(f"\nResponse:\n{state['response']}")

    return state

from langgraph.graph import END, StateGraph
from langchain_aws import ChatBedrockConverse
from langchain_core.messages import HumanMessage
from typing import TypedDict, Literal
from pydantic import BaseModel, Field
from rich import print
import re
class StateSchema(TypedDict):
    name: str
    content: str
    is_safe: bool
    llm_response: str
    toxicity_confidence: float
class ToxicityScore(BaseModel):
    toxicity: Literal['toxic', 'safe']
    confidence: float = Field(ge=0.0, le=1.0)
PII_PATTERNS = {
    'email': r"\b[a-zA-Z0-9._%+-]+@[a-zA-Z0-9.-]+\.[A-Za-z]{2,}\b",
    'phone': r"\b(?:\+?\d{1,3}[-.\s]?)?(?:\(?\d{3,4}\)?[-.\s]?)?\d{3,4}[-.\s]?\d{4}\b",
    'credit_card': r"\b(?:\d[ -]*?){13,16}\b",
    'ssn': r"\b[2-9]{1}[0-9]{3}[-\s]?[0-9]{4}[-\s]?[0-9]{4}\b",
    'pan': r"\b[A-Z]{5}[0-9]{4}[A-Z]\b"
}
def get_content(state: StateSchema) -> StateSchema:
    print(f"Hello {state['name']}! Let's collect the content.")
    raw = input("Enter details: ")
    return {"content": raw}
def validate_input(state: StateSchema) -> StateSchema:
    content = state["content"]
    detected_pii = [pii for pii, pattern in PII_PATTERNS.items() if re.search(pattern, content)]
    if detected_pii:
        pii_types = ", ".join(detected_pii)
        print(f"\nPII Detected: {pii_types}")
        return {
            "name": state["name"],
            "content": state["content"],
            "is_safe": False,
            "llm_response": f"Security Alert: Your input contains sensitive personal information ({pii_types}). Please remove it and try again.",
            "toxicity_confidence": 0.0
        }
    print("\nChecking for toxic content...")
    llm = ChatBedrockConverse(model="amazon.nova-lite-v1:0", temperature=0)
    structured_llm = llm.with_structured_output(ToxicityScore)
    prompt = f"Classify this text for toxicity: \"{content}\"\nRespond with 'toxicity' ('toxic' or 'safe') and 'confidence' (0.0-1.0)."
    try:
        result = structured_llm.invoke([HumanMessage(content=prompt)])
        toxicity = result.toxicity.lower()
        confidence = result.confidence
        print(f"Result: {toxicity} (confidence: {confidence:.2f})")
    except Exception:
        toxicity = "safe"
        confidence = 0.0
    if toxicity == "toxic" and confidence > 0.7:
        print(f"\nToxic content detected!")
        return {
            "name": state["name"],
            "content": state["content"],
            "is_safe": False,
            "llm_response": f"Content Policy Violation: Your input has been flagged as toxic (confidence: {confidence:.0%}). Please rephrase appropriately.",
            "toxicity_confidence": confidence
        }
    print(f"Content is safe")
    return {
        "name": state["name"],
        "content": state["content"],
        "is_safe": True,
        "llm_response": "",
        "toxicity_confidence": confidence
    }
def show_summary(state: StateSchema) -> StateSchema:
    print(f"\n{'='*60}\n{'SUMMARY':^60}\n{'='*60}")
    print(f"Name: {state['name']}")
    print(f"Input: {state['content']}")
    print(f"Safe: {'✅' if state['is_safe'] else '❌'}")
    print(f"Toxicity: {state['toxicity_confidence']:.0%}")

        print(f"\n{state['llm_response']}")
        return state

    response = llm.invoke([HumanMessage(content=f"Respond to: {state['content']}")])
    print(f"\n {response.content}")
    return {
        "name": state["name"],
        "content": state["content"],
        "is_safe": state["is_safe"],
        "llm_response": response.content,
        "toxicity_confidence": state["toxicity_confidence"]
    }

    print("="*60)
    while True:
        decision = input("\nDo you want to APPROVE or REJECT this content? (approve/reject): ").strip().lower()

            break
        print("Invalid input. Please enter 'approve' or 'reject'.")
    feedback = ""
    if decision == 'reject':
        feedback = input("Optional - Provide rejection reason: ").strip()
    return {
        "name": state["name"],
        "content": state["content"],
        "is_safe": state["is_safe"],
        "llm_response": state.get("llm_response", ""),

        "human_approved": decision == 'approve',

    }
